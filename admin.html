⚙️ Supabase Cloud Sync Setup (Optional SQL)

If you want real-time syncing across separate devices without permission barriers, run this once in your Supabase SQL Editor:

-- 1. Create table if it doesn't already exist
CREATE TABLE IF NOT EXISTS public.wedding_rsvps (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    guest_name TEXT NOT NULL,
    attending BOOLEAN DEFAULT TRUE,
    message TEXT,
    rating INTEGER
);

-- 2. Enable Row Level Security & allow guest submissions and admin viewing
ALTER TABLE public.wedding_rsvps ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Allow public insert" ON public.wedding_rsvps;
DROP POLICY IF EXISTS "Allow public select" ON public.wedding_rsvps;
DROP POLICY IF EXISTS "Allow public delete" ON public.wedding_rsvps;

CREATE POLICY "Allow public insert" ON public.wedding_rsvps FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public select" ON public.wedding_rsvps FOR SELECT USING (true);
CREATE POLICY "Allow public delete" ON public.wedding_rsvps FOR DELETE USING (true);
